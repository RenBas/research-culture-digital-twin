@dataclass
class CycleRecord:
    cycle_number: int
    total_improvement: float
    completion_month: int

class SchoolAgent:
    def __init__(self, unique_id,
                 initial_R=0.3, initial_A=0.2, initial_C=0.2,
                 initial_S=0.1, initial_I=0.1, initial_P=0.1, initial_M=0.0,
                 random_events_enabled=False):
        self.id = unique_id
        self.R = initial_R
        self.A = initial_A
        self.C = initial_C
        self.S = initial_S
        self.I = initial_I
        self.P = initial_P
        self.M = initial_M
        self.current_milestone = 0
        self.months_in_milestone = 0
        self.current_cycle_accumulator = 0.0
        self.cycle_improvements: List[CycleRecord] = []
        self.cycle_count = 0
        self.running_total_outcome = 0.0
        self.min_value = 0.1
        self.random_events_enabled = random_events_enabled
        self.model_time = 0
        self.random = np.random.RandomState()

    def apply_random_event(self):
        if not self.random_events_enabled:
            return
        if self.random.rand() < 0.00417:
            event_type = self.random.choice(["loss_champion", "funding", "leadership_change"])
            if event_type == "loss_champion":
                for var in ['R','A','C','S','I','P','M']:
                    setattr(self, var, max(self.min_value, getattr(self, var) - 0.1))
            elif event_type == "funding":
                self.S = min(1.0, self.S + 0.15)
            elif event_type == "leadership_change":
                self.I = max(self.min_value, self.I - 0.2)

    def step(self, levers: Dict[str, float]):
        u_train, u_mentor, u_budget, u_lead, u_collab = levers.values()
        u_lead_eff = min(1.0, u_lead + 0.05 * self.M)
        self.R += 0.02 * self.M - 0.01 * (1 - u_lead_eff)
        self.A += 0.04 * self.R + 0.02 * u_train + 0.01 * self.M - 0.005
        self.C += 0.03 * u_train + 0.02 * u_mentor - 0.01
        self.S += 0.04 * u_budget + 0.02 * u_mentor - 0.01 * (1 - u_lead_eff)
        self.I += 0.03 * u_lead_eff + 0.02 * self.S - 0.005
        self.P += 0.04 * u_collab + 0.02 * self.I - 0.01
        self.M += 0.02 * self.C + 0.02 * self.P - 0.005
        for var in ['R','A','C','S','I','P','M']:
            setattr(self, var, max(self.min_value, min(1.0, getattr(self, var))))
        monthly_gain = 0.001 * self.M * (1 + self.P)
        self.running_total_outcome += monthly_gain
        self.current_cycle_accumulator += monthly_gain
        self._update_milestone()
        self.apply_random_event()

    def _update_milestone(self):
        self.months_in_milestone += 1
        next_milestone = self.current_milestone
        if self.current_milestone == 0 and self.A >= 0.8:
            next_milestone = 1
        elif self.current_milestone == 1 and self.C >= 0.7:
            next_milestone = 2
        elif self.current_milestone == 2 and self.S >= 0.7:
            next_milestone = 3
        elif self.current_milestone == 3 and self.I >= 0.8:
            next_milestone = 4
        elif self.current_milestone == 4 and self.P >= 0.8:
            next_milestone = 5
        elif self.current_milestone == 5 and self.M >= 0.7:
            next_milestone = 6
        elif self.current_milestone == 6 and self.M >= 0.9 and self.R >= 0.8:
            self._complete_cycle()
            next_milestone = 0
        if next_milestone != self.current_milestone and self.months_in_milestone >= 6:
            self.current_milestone = next_milestone
            self.months_in_milestone = 0

    def _complete_cycle(self):
        old_M = self.M
        self.R = min(1.0, self.R + 0.10)
        self.M = max(0.2, self.M * 0.5)
        bonus = 0.03 + 0.03 * old_M
        self.running_total_outcome += bonus
        self.current_cycle_accumulator += bonus
        self.cycle_count += 1
        self.cycle_improvements.append(CycleRecord(cycle_number=self.cycle_count,
                                                   total_improvement=self.current_cycle_accumulator,
                                                   completion_month=self.model_time))
        self.current_cycle_accumulator = 0.0

class Simulation:
    def __init__(self, num_schools=1, random_events=False):
        self.agents = [SchoolAgent(i, random_events_enabled=random_events) for i in range(num_schools)]
    def step(self, levers, month):
        for agent in self.agents:
            agent.model_time = month
            agent.step(levers)
    def get_agent(self, idx=0):
        return self.agents[idx]

def load_drive_data():
    drive_path = "/content/drive/MyDrive/DigitalTwin_Data"
    survey_file = os.path.join(drive_path, "quarterly_survey_data.csv")
    meta_file = os.path.join(drive_path, "research_metadata.csv")
    if not os.path.exists(survey_file):
        survey_file = os.path.join(drive_path, "quarterly_survey_data.xlsx")
    if not os.path.exists(meta_file):
        meta_file = os.path.join(drive_path, "research_metadata.xlsx")
    if os.path.exists(survey_file) and os.path.exists(meta_file):
        print(f"Loading survey from {survey_file}")
        print(f"Loading metadata from {meta_file}")
        if survey_file.endswith('.csv'):
            survey_df = pd.read_csv(survey_file)
        else:
            survey_df = pd.read_excel(survey_file, engine='openpyxl')
        if meta_file.endswith('.csv'):
            metadata_df = pd.read_csv(meta_file)
        else:
            metadata_df = pd.read_excel(meta_file, engine='openpyxl')
        # Ensure required columns
        if 'school_id_no' not in survey_df.columns:
            if 'school_id' in survey_df.columns:
                survey_df['school_id_no'] = survey_df['school_id'].astype(str).apply(lambda x: int(x.split('_')[-1]) if '_' in x else int(x))
            else:
                raise ValueError("Survey file must contain 'school_id_no' or 'school_id'")
        if 'school_name' not in survey_df.columns:
            survey_df['school_name'] = survey_df['school_id_no'].apply(lambda x: f"School_{x}")
        if 'school_id_no' not in metadata_df.columns:
            if 'school' in metadata_df.columns:
                metadata_df['school_id_no'] = metadata_df['school'].astype(str).apply(lambda x: int(x.split('_')[-1]) if '_' in x else int(x))
            else:
                raise ValueError("Metadata file must contain 'school_id_no' or 'school'")
        if 'year_undertaken' not in metadata_df.columns:
            print("Adding missing 'year_undertaken' column with default 2025.")
            metadata_df['year_undertaken'] = 2025
        if 'utilization_date' not in metadata_df.columns:
            metadata_df['utilization_date'] = ''
        if 'publication_link' not in metadata_df.columns:
            metadata_df['publication_link'] = ''
        def month_str_to_num(month_str):
            try:
                year = int(month_str[:4])
                month = int(month_str[5:])
                return (year - 2026) * 12 + month
            except:
                return 0
        survey_df['month_num'] = survey_df['month'].apply(month_str_to_num)
        return survey_df, metadata_df
    else:
        print("No Drive files found. Generating internal mock data for 20 schools (max slider will be 20).")
        schools = [i+1 for i in range(20)]
        quarters = [f"2026-{str(q).zfill(2)}" for q in [1,4,7,10]]
        rows = []
        for sid in schools:
            for q in quarters:
                R = np.random.uniform(0.2,0.4)
                A = np.random.uniform(0.2,0.5)
                C = np.random.uniform(0.1,0.3)
                S = np.random.uniform(0.1,0.3)
                I = np.random.uniform(0.1,0.2)
                P = np.random.uniform(0.1,0.2)
                M = np.random.uniform(0.0,0.1)
                rows.append([q, sid, f"School_{sid}", R, A, C, S, I, P, M])
        survey_df = pd.DataFrame(rows, columns=["month","school_id_no","school_name","R","A","C","S","I","P","M"])
        survey_df['month_num'] = survey_df['month'].apply(lambda x: (int(x[:4])-2026)*12 + int(x[5:]))
        meta_rows = []
        for sid in schools:
            for j in range(1, 4):
                teacher = f"Teacher_{sid}_{j}"
                meta_rows.append([f"2025-0{1+j}", teacher, sid, "full_paper", f"Research {j}", "Teaching Strategies", "published", "https://doi.org/10.1234", True, "2025-02-10", 2024+j])
        metadata_df = pd.DataFrame(meta_rows, columns=["upload_date","teacher_name","school_id_no","document_type","title","theme","status","publication_link","utilized_by_school","utilization_date","year_undertaken"])
        metadata_df['upload_date'] = pd.to_datetime(metadata_df['upload_date'])
        return survey_df, metadata_df

# 3. DASHBOARD WITH AUTO-RESEARCH UPDATE & MAX 200 SCHOOLS
# ------------------------------------------------------------
def build_dashboard(survey_df, metadata_df):
    school_info = survey_df[['school_id_no', 'school_name']].drop_duplicates().sort_values('school_id_no')
    school_list = [(f"ID {row.school_id_no}: {row.school_name}", row.school_id_no) for _, row in school_info.iterrows()]
    school_id_to_name = dict(zip(school_info['school_id_no'], school_info['school_name']))
    num_available_schools = len(school_info)

    print(f"Loaded {num_available_schools} schools.")
    print("Survey data (first 2 rows):")
    print(survey_df.head(2))
    print("\nResearch Metadata (first 3 rows):")
    print(metadata_df.head(3))

    # Widgets
    style = {'description_width': 'initial'}
    u_train_slider = widgets.FloatSlider(value=0.5, min=0, max=1, step=0.05, description='Training freq.', style=style)
    u_mentor_slider = widgets.FloatSlider(value=0.5, min=0, max=1, step=0.05, description='Mentorship ratio', style=style)
    u_budget_slider = widgets.FloatSlider(value=0.5, min=0, max=1, step=0.05, description='Support budget', style=style)
    u_lead_slider = widgets.FloatSlider(value=0.5, min=0, max=1, step=0.05, description='Leadership commit.', style=style)
    u_collab_slider = widgets.FloatSlider(value=0.5, min=0, max=1, step=0.05, description='Collaboration freq.', style=style)
    # Max schools set to 200, but limited by actual data
    num_schools_slider = widgets.IntSlider(value=min(20, num_available_schools), min=1, max=min(200, num_available_schools), step=1, description='Number of schools', style=style)
    random_events_check = widgets.Checkbox(value=False, description='Enable random events')
    use_survey_check = widgets.Checkbox(value=True, description='Override with survey data')
    school_selector = widgets.Dropdown(options=school_list, value=school_list[0][1], description='Select school:', style=style)
    
    duration_options = [12, 24, 36, 48, 60, 72, 84, 96, 108, 120]
    duration_dropdown = widgets.Dropdown(options=duration_options, value=120, description='Run duration (months):', style=style)
    
    run_button = widgets.Button(description='Run')
    step_button = widgets.Button(description='Step (1 month)')
    reset_button = widgets.Button(description='Reset')
    export_button = widgets.Button(description='Export results (CSV)')
    show_research_button = widgets.Button(description='Show Research Outputs')
    research_output = widgets.Output()

    out = widgets.Output()
    sim = None
    history_per_school = {}
    current_month = 0
    total_months = 0
    last_selected_school = None

    # Outcome interpretation table
    outcome_intervals = [
        (0.0, 0.2, "Very Low", "Little to no improvement in student learning outcomes."),
        (0.2, 0.4, "Low", "Minimal improvement; research culture still weak."),
        (0.4, 0.6, "Moderate", "Noticeable improvement; research culture developing."),
        (0.6, 0.8, "High", "Strong improvement; research culture becoming sustainable."),
        (0.8, 1.0, "Very High", "Excellent improvement; research culture fully embedded and impactful.")
    ]

    def interpret_outcome(value):
        for low, high, label, desc in outcome_intervals:
            if low <= value < high:
                return label, desc
        return "Exceptional", "Beyond expected range; outstanding impact."

    def generate_outcome_table():
        table_html = """
        <div style="margin-top: 10px; padding: 8px; border: 1px solid #ccc; border-radius: 5px;">
        <b>📈 Cumulative Student Outcome Interpretation Table:</b>
        <table style="width:100%; border-collapse: collapse;">
        <tr style="background-color:#ddd;"><th>Range</th><th>Level</th><th>Description</th></tr>
        """
        for low, high, label, desc in outcome_intervals:
            table_html += f"<tr><td>{low:.1f} – {high:.1f}</td><td>{label}</td><td>{desc}</td></tr>"
        table_html += "</table></div>"
        return HTML(table_html)

    def update_research_table():
        with research_output:
            clear_output(wait=True)
            selected_id = school_selector.value
            if selected_id is None:
                print("No school selected.")
                return
            df_show = metadata_df[metadata_df['school_id_no'] == selected_id].copy()
            if df_show.empty:
                print(f"No research outputs for School ID {selected_id}.")
                return
            df_show_sorted = df_show.sort_values('upload_date')
            df_show_sorted['cumulative_by_teacher'] = df_show_sorted.groupby('teacher_name').cumcount() + 1
            display_cols = ['teacher_name', 'year_undertaken', 'title', 'theme', 'status', 'cumulative_by_teacher']
            print(f"Research outputs for School ID {selected_id} ({school_id_to_name[selected_id]}):")
            display(df_show_sorted[display_cols])
            total_per_teacher = df_show_sorted.groupby('teacher_name').size()
            print("\nTotal per teacher:\n", total_per_teacher)
            print(f"\nOverall total: {len(df_show_sorted)}")

    def initialize_simulation(reset=True):
        nonlocal sim, history_per_school, current_month, total_months, last_selected_school
        if reset:
            num = num_schools_slider.value
            selected_ids = school_info['school_id_no'].head(num).tolist()
            sim = Simulation(num_schools=num, random_events=random_events_check.value)
            for idx, agent in enumerate(sim.agents):
                agent.real_id = selected_ids[idx]
            for agent in sim.agents:
                school_metadata = metadata_df[metadata_df['school_id_no'] == agent.real_id]
                agent.A = min(1.0, agent.A + len(school_metadata[school_metadata['document_type']=='abstract'])*0.01)
                agent.M = min(1.0, agent.M + len(school_metadata[school_metadata['status']=='published'])*0.02)
                agent.C = min(1.0, agent.C + len(school_metadata[school_metadata['document_type']=='full_paper'])*0.005)
                agent.P = min(1.0, agent.P + school_metadata['theme'].nunique()*0.01)
            history_per_school = {}
            for agent in sim.agents:
                history_per_school[agent.real_id] = {k:[] for k in ['R','A','C','S','I','P','M','month','milestone','running_outcome']}
            current_month = 0
            total_months = 0
            present_ids = [agent.real_id for agent in sim.agents]
            present_options = [(f"ID {sid}: {school_id_to_name[sid]}", sid) for sid in present_ids]
            school_selector.options = present_options
            if last_selected_school is not None and last_selected_school in present_ids:
                school_selector.value = last_selected_school
            elif present_options:
                school_selector.value = present_options[0][1]
            else:
                school_selector.value = None
            # After reset, update research outputs for the new selection
            update_research_table()

    def generate_synopses():
        if not sim or total_months == 0:
            return "", ""
        selected_id = school_selector.value
        if selected_id is None:
            return "", ""
        agent = next((a for a in sim.agents if a.real_id == selected_id), None)
        if agent:
            outcome_val = agent.running_total_outcome
            level, desc = interpret_outcome(outcome_val)
            per_school_text = f"""
            <div style="background-color: #e8f0fe; padding: 8px; border-radius: 5px; margin-bottom: 8px;">
            <b>📌 School {selected_id} ({school_id_to_name[selected_id]}) Synopsis:</b><br>
            After {total_months} months: Milestone = {agent.current_milestone} | Completed cycles = {agent.cycle_count}<br>
            Cumulative student outcome improvement = <b>{outcome_val:.3f}</b> → <b>{level}</b> level: {desc}<br>
            <i>Interpretation:</i> {
                "The school has reached a self‑sustaining research culture (multiple cycles)." if agent.cycle_count >= 2 else
                "The school is building its research culture but has not yet completed a full cycle." if agent.cycle_count == 0 else
                "The school has completed one full cycle, showing initial sustainability."
            }
            </div>
            """
        else:
            per_school_text = "<div>No data for selected school.</div>"
        
        total_outcome = sum(a.running_total_outcome for a in sim.agents)
        avg_outcome = total_outcome / len(sim.agents)
        level_avg, desc_avg = interpret_outcome(avg_outcome)
        total_cycles = sum(a.cycle_count for a in sim.agents)
        avg_milestone = np.mean([a.current_milestone for a in sim.agents])
        div_text = f"""
        <div style="background-color: #f0f0f0; padding: 8px; border-radius: 5px;">
        <b>🏢 Division‑Level Synopsis (all {len(sim.agents)} schools):</b><br>
        Average milestone = {avg_milestone:.1f} | Total completed cycles across all schools = {total_cycles}<br>
        Cumulative student outcome improvement (sum) = {total_outcome:.3f}, average per school = {avg_outcome:.3f} → <b>{level_avg}</b> level: {desc_avg}<br>
        <i>Division‑wide interpretation:</i> {
            "The division is showing strong research culture with multiple cycles and high impact." if total_cycles > len(sim.agents) else
            "Most schools are still in early stages of research culture development." if avg_milestone < 3 else
            "The division has a moderate research culture; policy adjustments may accelerate progress."
        }
        </div>
        """
        return per_school_text, div_text

    interpretations = HTML("""
    <div style="background-color: #f0f0f0; padding: 10px; border-radius: 5px; margin-top: 10px;">
    <h4>📊 Graph Interpretations</h4>
    <ul>
    <li><b>Variable Evolution:</b> Shows how the seven research culture components (R, A, C, S, I, P, M) change over time. Higher values (closer to 1) mean stronger readiness, awareness, capacity, etc. The goal is to see all lines trend upward.</li>
    <li><b>Milestone Progress:</b> The school moves through milestones 0–6. Reaching milestone 6 (Impact Realization) and then cycling back to 0 indicates a full sustainable cycle. Faster progression means better policy effectiveness.</li>
    <li><b>Student Learning Outcome (Running Total):</b> Cumulative improvement in learner outcomes attributed to the research culture. The steeper the curve, the greater the impact.</li>
    <li><b>Improvement per Completed Cycle:</b> Each bar represents the total improvement in student outcomes contributed by one complete cycle. Higher bars in later cycles show that the research ecosystem becomes more effective over time.</li>
    </ul>
    <p><i>Use sliders to adjust policies (e.g., training, budget, leadership) and observe how these graphs change. Select duration and click "Run".</i></p>
    </div>
    """)

    def update_plots():
        with out:
            clear_output(wait=True)
            if not sim or total_months == 0:
                print("No data yet. Select duration and click 'Run' or 'Step'.")
                display(interpretations)
                display(generate_outcome_table())
                return
            selected_id = school_selector.value
            if selected_id is None:
                print("No school selected.")
                display(interpretations)
                display(generate_outcome_table())
                return
            if selected_id not in history_per_school:
                print(f"No history for school ID {selected_id}. Run simulation first.")
                display(interpretations)
                display(generate_outcome_table())
                return
            hist = history_per_school[selected_id]
            agent = next((a for a in sim.agents if a.real_id == selected_id), None)
            if agent is None:
                print("Agent not found.")
                return

            fig, axes = plt.subplots(2, 2, figsize=(14, 8))
            for var in ['R','A','C','S','I','P','M']:
                axes[0,0].plot(hist['month'], hist[var], label=var)
            axes[0,0].set_title(f'School ID {selected_id}: Variable Evolution')
            axes[0,0].set_xlabel('Month')
            axes[0,0].set_ylabel('Value (0-1)')
            axes[0,0].legend(loc='upper left', fontsize=8)
            axes[0,0].grid(True)
            axes[0,1].step(hist['month'], hist['milestone'], where='post', color='red')
            axes[0,1].set_title('Milestone Progress')
            axes[0,1].set_xlabel('Month')
            axes[0,1].set_ylabel('Milestone')
            axes[0,1].set_ylim(-0.5,6.5)
            axes[0,1].grid(True)
            axes[1,0].plot(hist['month'], hist['running_outcome'], 'g-')
            axes[1,0].set_title('Student Learning Outcome (Running Total)')
            axes[1,0].set_xlabel('Month')
            axes[1,0].set_ylabel('Cumulative Improvement')
            axes[1,0].grid(True)
            if agent.cycle_improvements:
                cycles = [c.cycle_number for c in agent.cycle_improvements]
                improvements = [c.total_improvement for c in agent.cycle_improvements]
                axes[1,1].bar(cycles, improvements, color='orange')
                axes[1,1].set_title('Improvement per Completed Cycle')
                axes[1,1].set_xlabel('Cycle Number')
                axes[1,1].set_ylabel('Improvement')
            else:
                axes[1,1].text(0.5,0.5,'No cycles completed yet', ha='center', va='center')
                axes[1,1].set_title('Per-Cycle Improvement')
            plt.tight_layout()
            plt.show()
            display(interpretations)
            display(generate_outcome_table())
            per_school_syn, div_syn = generate_synopses()
            display(HTML(per_school_syn + div_syn))
            print(f"School ID {selected_id} ({school_id_to_name[selected_id]}) | Month: {total_months} | Milestone: {agent.current_milestone} | Cycles: {agent.cycle_count} | Running Outcome: {agent.running_total_outcome:.3f}")

    def run_simulation(months, reset_first=False):
        nonlocal sim, current_month, total_months, last_selected_school
        last_selected_school = school_selector.value
        if reset_first:
            initialize_simulation(reset=True)
        if sim is None:
            initialize_simulation(reset=True)
        levers = {k: v.value for k, v in zip(['u_train','u_mentor','u_budget','u_lead','u_collab'],
                                             [u_train_slider, u_mentor_slider, u_budget_slider, u_lead_slider, u_collab_slider])}
        for m in range(1, months+1):
            if use_survey_check.value:
                for agent in sim.agents:
                    row = survey_df[(survey_df['school_id_no'] == agent.real_id) & (survey_df['month_num'] == current_month + m)]
                    if not row.empty:
                        r = row.iloc[0]
                        agent.R, agent.A, agent.C, agent.S, agent.I, agent.P, agent.M = r[['R','A','C','S','I','P','M']]
            sim.step(levers, current_month + m)
            current_month += 1
            total_months += 1
            for agent in sim.agents:
                h = history_per_school[agent.real_id]
                h['month'].append(total_months)
                for var in ['R','A','C','S','I','P','M']:
                    h[var].append(getattr(agent, var))
                h['milestone'].append(agent.current_milestone)
                h['running_outcome'].append(agent.running_total_outcome)
        update_plots()

    def on_run(b): 
        months = duration_dropdown.value
        run_simulation(months, reset_first=True)
    def on_step(b): run_simulation(1, reset_first=False)
    def on_reset(b): 
        global last_selected_school
        last_selected_school = school_selector.value
        run_simulation(0, reset_first=True)

                for var in ['R','A','C','S','I','P','M']:
                    row[var] = h[var][t]
                all_data.append(row)
